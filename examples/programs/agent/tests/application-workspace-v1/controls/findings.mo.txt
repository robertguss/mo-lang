module Agent.Tests.Red.Findings

use Agent.WorkspaceAdapter{Settings, checked, local, stops?}

intent "Carried review findings 1 to 4 as controls over the adapter's response check; materialized outside the corpus."

fn configured() : Settings
  Settings(port: 1, run: "r", workspace: "0123456789abcdef0123456789abcdef", token: "ab".repeat(32))
end

fn unknown() : String
  local("failure", "invalid_response", "unknown")
end

test "finding 1: a pre-admission refusal with null identities keeps not_started"
  body = "{\"version\":\"mo-workspace-http-v1\",\"run_id\":null,\"workspace_id\":null,\"call_id\":null,\"accepted\":false,\"state\":\"refusal\",\"execution\":\"not_started\",\"error\":\"unauthorized\",\"result\":null}"
  kept = checked(401, body, configured(), "1", "read_file")
  assert kept == body
  assert stops?(kept)
end

test "finding 2: a busy refusal on HTTP 500 violates the contract, since busy is 409"
  body = "{\"version\":\"mo-workspace-http-v1\",\"run_id\":\"r\",\"workspace_id\":\"0123456789abcdef0123456789abcdef\",\"call_id\":\"1\",\"accepted\":false,\"state\":\"refusal\",\"execution\":\"not_started\",\"error\":\"busy\",\"result\":null}"
  assert checked(500, body, configured(), "1", "read_file") == unknown()
  assert checked(409, body, configured(), "1", "read_file") == body
end

test "finding 3: command streams are both strings, or both null with failure/output_encoding"
  head = "{\"version\":\"mo-workspace-http-v1\",\"run_id\":\"r\",\"workspace_id\":\"0123456789abcdef0123456789abcdef\",\"call_id\":\"1\",\"accepted\":true,"
  success_null = "#{head}\"state\":\"success\",\"execution\":\"completed\",\"error\":null,\"result\":{\"exit_code\":0,\"signal\":null,\"stdout\":null,\"stderr\":null,\"encoding\":\"utf-8\",\"execution_valid\":true,\"truncated\":false,\"elapsed_ms\":0}}"
  assert checked(200, success_null, configured(), "1", "command") == unknown()
  mixed = "#{head}\"state\":\"failure\",\"execution\":\"completed\",\"error\":\"output_encoding\",\"result\":{\"exit_code\":0,\"signal\":null,\"stdout\":\"\",\"stderr\":null,\"encoding\":\"utf-8\",\"execution_valid\":true,\"truncated\":false,\"elapsed_ms\":0}}"
  assert checked(200, mixed, configured(), "1", "command") == unknown()
  swapped = mixed.replace("\"stdout\":\"\",\"stderr\":null", "\"stdout\":null,\"stderr\":\"\"")
  assert checked(200, swapped, configured(), "1", "command") == unknown()
  both_null = mixed.replace("\"stdout\":\"\",", "\"stdout\":null,")
  assert checked(200, both_null, configured(), "1", "command") == both_null
end

test "finding 4: success with a nonzero exit code is rejected"
  body = "{\"version\":\"mo-workspace-http-v1\",\"run_id\":\"r\",\"workspace_id\":\"0123456789abcdef0123456789abcdef\",\"call_id\":\"1\",\"accepted\":true,\"state\":\"success\",\"execution\":\"completed\",\"error\":null,\"result\":{\"exit_code\":1,\"signal\":null,\"stdout\":\"\",\"stderr\":\"\",\"encoding\":\"utf-8\",\"execution_valid\":true,\"truncated\":false,\"elapsed_ms\":0}}"
  assert checked(200, body, configured(), "1", "command") == unknown()
  zero = body.replace("\"exit_code\":1", "\"exit_code\":0")
  assert checked(200, zero, configured(), "1", "command") == zero
  signalled = zero.replace("\"signal\":null", "\"signal\":9")
  assert checked(200, signalled, configured(), "1", "command") == signalled
end
